# Build the C++/Qt editor and its headless document and IPC verification suite.
cmake_minimum_required(VERSION 3.25)
project(DarkWinterBiomeEditor VERSION 0.1.0 LANGUAGES CXX)
include(cmake/MsvcDependencyPrefix.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(DARKWINTER_VULKAN_HEADERS "" CACHE PATH "Pinned Vulkan-Headers include directory")
if(DARKWINTER_VULKAN_HEADERS AND NOT Vulkan_INCLUDE_DIR)
    set(Vulkan_INCLUDE_DIR "${DARKWINTER_VULKAN_HEADERS}" CACHE PATH "Vulkan headers" FORCE)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT Vulkan_INCLUDE_DIR)
    message(FATAL_ERROR "Set Vulkan_INCLUDE_DIR or DARKWINTER_VULKAN_HEADERS to the Vulkan-Headers include directory")
endif()
add_library(editor_core STATIC src/Document.cpp src/Document.hpp src/WorkerClient.cpp src/WorkerClient.hpp src/SceneBundle.cpp src/SceneBundle.hpp)
target_include_directories(editor_core PUBLIC src)
target_include_directories(editor_core SYSTEM PUBLIC "${Vulkan_INCLUDE_DIR}")
target_link_libraries(editor_core PUBLIC Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(editor_core PUBLIC /W4 /permissive- /EHsc /utf-8)
    target_compile_definitions(editor_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(DARKWINTER_BUILD_EDITOR "Build the Vulkan desktop application" ON)
if(DARKWINTER_BUILD_EDITOR)
    qt_add_executable(darkwinter-biome-editor WIN32
        src/main.cpp src/EditorWindow.cpp src/EditorWindow.hpp src/EditorSmoke.cpp src/EditorSceneSmoke.cpp src/EditorWorkerSmoke.cpp
        src/PreviewViewport.cpp src/PreviewViewport.hpp)
    target_link_libraries(darkwinter-biome-editor PRIVATE editor_core Qt6::Widgets)
    set_target_properties(darkwinter-biome-editor PROPERTIES OUTPUT_NAME "DarkWinterBiomeEditor")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(editor_ipc_fixture tests/ipc_worker_fixture.cpp)
    target_link_libraries(editor_ipc_fixture PRIVATE Qt6::Core)
    add_executable(editor_core_test tests/editor_core_test.cpp)
    target_link_libraries(editor_core_test PRIVATE editor_core)
    add_test(NAME editor_document_and_ipc COMMAND editor_core_test $<TARGET_FILE:editor_ipc_fixture>)
    add_executable(editor_preview_math_test tests/preview_math_test.cpp)
    target_link_libraries(editor_preview_math_test PRIVATE Qt6::Core)
    add_test(NAME editor_preview_math COMMAND editor_preview_math_test)
    get_target_property(_qtCoreDll Qt6::Core IMPORTED_LOCATION_RELEASE)
    get_filename_component(_qtBin "${_qtCoreDll}" DIRECTORY)
    set_tests_properties(editor_document_and_ipc editor_preview_math PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtBin}")
endif()
