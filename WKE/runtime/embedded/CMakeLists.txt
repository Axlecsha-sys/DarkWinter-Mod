# Embedded WKE renderer borrows a host Vulkan device and never creates a window.
set(WKE_VULKAN_INCLUDE_DIR "" CACHE PATH "Directory containing vulkan/vulkan_core.h")
if(NOT EXISTS "${WKE_VULKAN_INCLUDE_DIR}/vulkan/vulkan_core.h")
    message(FATAL_ERROR "WKE embedded renderer requires Vulkan headers via WKE_VULKAN_INCLUDE_DIR")
endif()
find_package(Threads REQUIRED)
add_library(wke_render SHARED wke_vulkan.cpp material_mips.cpp)
add_library(WKE::EmbeddedRenderer ALIAS wke_render)
add_executable(wke_render_checks wke_vulkan.cpp wke_vulkan_checks.cpp)
add_executable(wke_material_checks material_mips.cpp material_mips_checks.cpp)
foreach(target IN ITEMS wke_render wke_render_checks wke_material_checks)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${WKE_VULKAN_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    endif()
endforeach()
add_test(NAME wke_render_native_contracts COMMAND wke_render_checks)
add_test(NAME wke_material_native_contracts COMMAND wke_material_checks)
