cmake_minimum_required(VERSION 3.20)
project(WinterKoreCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
set(CORE_SOURCES src/core.cpp src/format.cpp)
add_library(dw_core SHARED ${CORE_SOURCES})
target_link_libraries(dw_core PRIVATE Threads::Threads)
add_executable(dw_core_tool ${CORE_SOURCES} src/cli.cpp)
target_link_libraries(dw_core_tool PRIVATE Threads::Threads)
add_executable(dw_core_checks ${CORE_SOURCES} checks/cpp/checks.cpp)
target_link_libraries(dw_core_checks PRIVATE Threads::Threads)
enable_testing()
add_test(NAME core_checks COMMAND dw_core_checks ${CMAKE_BINARY_DIR}/check-worlds)
